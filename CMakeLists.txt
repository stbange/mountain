cmake_minimum_required(VERSION 3.24)
project(Mountain VERSION 0.1.0 LANGUAGES C)
include(FetchContent)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(MOUNTAIN_COVERAGE "Instrument Mountain sources for gcov" OFF)
option(MOUNTAIN_TESTS "Build the native test programs" ON)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SDL URL https://www.libsdl.org/release/SDL3-3.2.24.tar.gz
  URL_HASH SHA256=81cc0fc17e5bf2c1754eeca9af9c47a76789ac5efdd165b3b91cbbe4b90bfb76 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_Declare(lua URL https://www.lua.org/ftp/lua-5.4.8.tar.gz
  URL_HASH SHA256=4f18ddae154e793e46eeab727c59ef1c0c0c2b744e7b94219710d76f530629ae DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(SDL lua)
file(GLOB lua_sources CONFIGURE_DEPENDS "${lua_SOURCE_DIR}/src/*.c")
list(REMOVE_ITEM lua_sources "${lua_SOURCE_DIR}/src/lua.c" "${lua_SOURCE_DIR}/src/luac.c" "${lua_SOURCE_DIR}/src/onelua.c")
add_library(mountain_lua STATIC ${lua_sources})
target_include_directories(mountain_lua PUBLIC "${lua_SOURCE_DIR}/src")
add_library(mountain_engine STATIC src/world.c src/lighting.c src/realms.c src/expedition.c src/landmarks.c src/backcountry.c src/forest.c src/fauna.c src/survey.c src/route.c src/game.c src/cabin.c src/lookout.c src/kitchen.c src/script.c src/scene.c src/mesh.c)
target_include_directories(mountain_engine PUBLIC src)
target_link_libraries(mountain_engine PUBLIC mountain_lua)
if(NOT MSVC)
  target_link_libraries(mountain_engine PUBLIC m)
endif()
add_library(mountain_display STATIC src/render.c)
target_link_libraries(mountain_display PUBLIC mountain_engine SDL3::SDL3)
add_library(mountain_app STATIC src/app.c)
target_link_libraries(mountain_app PUBLIC mountain_display)
add_executable(mountain src/main.c)
target_link_libraries(mountain PRIVATE mountain_app)
set_target_properties(mountain PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/play")
if(MOUNTAIN_COVERAGE)
  foreach(t mountain_engine mountain_display mountain_app mountain)
    target_compile_options(${t} PRIVATE -O0 -g --coverage)
    target_link_options(${t} INTERFACE --coverage)
  endforeach()
  target_link_options(mountain PRIVATE --coverage)
endif()
if(EMSCRIPTEN)
  set_target_properties(mountain PROPERTIES SUFFIX ".js")
  target_link_options(mountain PRIVATE -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=33554432
    -sMAX_WEBGL_VERSION=2 -sMIN_WEBGL_VERSION=2 -sGL_ENABLE_GET_PROC_ADDRESS=1
    -sFORCE_FILESYSTEM=1 -sEXPORTED_RUNTIME_METHODS=FS,ccall -sEXPORTED_FUNCTIONS=_main,_malloc,_free
    --pre-js "${CMAKE_CURRENT_SOURCE_DIR}/web/preload.js")
  set_property(TARGET mountain APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/web/preload.js")
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_custom_target(mountain_assets ALL
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/stage.py"
          "${CMAKE_CURRENT_SOURCE_DIR}" "$<TARGET_FILE_DIR:mountain>" "${CMAKE_SYSTEM_NAME}"
  DEPENDS mountain VERBATIM)
if(MOUNTAIN_TESTS AND NOT EMSCRIPTEN)
  enable_testing()
  add_executable(mountain_lua_test "${lua_SOURCE_DIR}/src/lua.c")
  target_link_libraries(mountain_lua_test PRIVATE mountain_lua)
  if(NOT MSVC)
    target_link_libraries(mountain_lua_test PRIVATE m)
  endif()
  add_test(NAME lua-content COMMAND mountain_lua_test "${CMAKE_CURRENT_SOURCE_DIR}/tests/lua_coverage.lua" "${CMAKE_CURRENT_SOURCE_DIR}/assets")
  add_executable(mountain_unit tests/unit.c)
  target_link_libraries(mountain_unit PRIVATE mountain_app)
  add_test(NAME unit COMMAND mountain_unit "${CMAKE_CURRENT_SOURCE_DIR}/assets")
  add_executable(mountain_playtest tests/play.c)
  target_link_libraries(mountain_playtest PRIVATE mountain_app)
  add_test(NAME game COMMAND mountain_playtest "${CMAKE_CURRENT_SOURCE_DIR}/assets")
endif()
